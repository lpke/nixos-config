cmake_minimum_required(VERSION 3.22)
project(scriptrecovery LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Quick DBus)
include(KDEInstallDirs6)
find_package(KWin 6.6 REQUIRED)
add_library(scriptrecovery MODULE recovery.cpp)
set_target_properties(scriptrecovery PROPERTIES PREFIX "")
target_link_libraries(scriptrecovery PRIVATE KWin::kwin)
install(TARGETS scriptrecovery DESTINATION ${KDE_INSTALL_PLUGINDIR}/kwin/effects/plugins)
